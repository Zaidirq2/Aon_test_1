experiment2.m Experiment 2: Digital Modulation Techniques (BPSK, QPSK, 16-QAM)

1%% Experiment 2: Digital Modulation Techniques
2clc; clear; close all;
3rng(1);                              % Same results every run
4
5N    = 120000;                       % Divisible by 1, 2 and 4 bits/symbol
6data = randi([0 1], N, 1);           % Random binary data
7
8schemes = {'BPSK', 'QPSK', '16-QAM'};
9M_list  = [2 4 16];
10tx      = cell(1, 3);
11
12%% Parts 1-3: Modulation order, bits/symbol, ideal constellations
13figure('Name', 'Ideal Constellations');
14for s = 1:3
15    M = M_list(s);
16    k = log2(M);                     % Bits per symbol
17    tx{s} = modulateBits(data, M);
18
19    fprintf('%-7s: M = %2d,  k = log2(M) = %d bits/symbol,  symbols sent = %d\n', ...
20            schemes{s}, M, k, N/k);
21
22    % Ideal points with their bit labels
23    labels = dec2bin(0:M-1, k);
24    ideal  = modulateBits(reshape((labels - '0').', [], 1), M);
25
26    subplot(1, 3, s);
27    plot(real(ideal), imag(ideal), 'bo', 'MarkerFaceColor', 'b'); hold on;
28    text(real(ideal) + 0.05, imag(ideal) + 0.12, cellstr(labels), 'Color', 'r', 'FontSize', 8);
29    grid on; axis equal; axis([-1.5 1.5 -1.5 1.5]);
30    xlabel('In-phase (I)'); ylabel('Quadrature (Q)');
31    title(sprintf('%s (M = %d, k = %d)', schemes{s}, M, k));
32end
33
34%% Part 4: BER vs SNR for all three schemes
35SNR_dB = 0:1:20;
36BER    = zeros(3, length(SNR_dB));
37BER_th = zeros(3, length(SNR_dB));
38
39for s = 1:3
40    M = M_list(s);
41    k = log2(M);
42    for i = 1:length(SNR_dB)
43        rxSig  = awgn(tx{s}, SNR_dB(i), 'measured');   % AWGN channel
44        rxData = demodulateBits(rxSig, M);             % Demodulate
45        [~, BER(s, i)] = biterr(data, rxData);         % Compare & compute BER
46    end
47    % Theoretical BER (awgn SNR is per symbol, so Eb/N0 = SNR - 10log10(k))
48    EbNo = SNR_dB - 10*log10(k);
49    if M == 16
50        BER_th(s, :) = berawgn(EbNo, 'qam', M);
51    else
52        BER_th(s, :) = berawgn(EbNo, 'psk', M, 'nondiff');
53    end
54end
55
56figure('Name', 'BER vs SNR');
57colors = {'b', 'g', 'r'};
58for s = 1:3
59    semilogy(SNR_dB, BER(s, :), [colors{s} 'o-'], 'LineWidth', 1.5); hold on;
60    semilogy(SNR_dB, BER_th(s, :), [colors{s} '--'], 'LineWidth', 1);
61end
62grid on; ylim([1e-6 1]);
63xlabel('SNR (dB)'); ylabel('BER');
64title('BER vs SNR: BPSK, QPSK, 16-QAM over AWGN');
65legend('BPSK sim', 'BPSK theory', 'QPSK sim', 'QPSK theory', ...
66       '16-QAM sim', '16-QAM theory', 'Location', 'southwest');
67
68% Comparison at the same SNR
69SNR_cmp = 10;
70idx = find(SNR_dB == SNR_cmp);
71fprintf('\nBER at SNR = %d dB:\n', SNR_cmp);
72for s = 1:3
73    fprintf('  %-7s: %g\n', schemes{s}, BER(s, idx));
74end
75
76%% Part 5: Received constellations at low and high SNR
77SNR_low  = 5;
78SNR_high = 20;
79nPlot    = 3000;
80
81figure('Name', 'Received Constellations');
82for s = 1:3
83    M  = M_list(s);
84    rL = awgn(tx{s}, SNR_low,  'measured');
85    rH = awgn(tx{s}, SNR_high, 'measured');
86    nP = min(nPlot, length(tx{s}));
87
88    subplot(2, 3, s);
89    plot(real(rL(1:nP)), imag(rL(1:nP)), 'b.', 'MarkerSize', 3); hold on;
90    plot(real(tx{s}), imag(tx{s}), 'r*', 'MarkerSize', 8);
91    grid on; axis equal; axis([-2 2 -2 2]);
92    title(sprintf('%s, SNR = %d dB', schemes{s}, SNR_low));
93    xlabel('I'); ylabel('Q');
94
95    subplot(2, 3, s + 3);
96    plot(real(rH(1:nP)), imag(rH(1:nP)), 'b.', 'MarkerSize', 3); hold on;
97    plot(real(tx{s}), imag(tx{s}), 'r*', 'MarkerSize', 8);
98    grid on; axis equal; axis([-2 2 -2 2]);
99    title(sprintf('%s, SNR = %d dB', schemes{s}, SNR_high));
100    xlabel('I'); ylabel('Q');
101end
102
103%% Part 6: Modulation order vs data rate (fixed symbol rate)
104Rs = 1e6;                            % Symbol rate: 1 Msymbol/s (assumed)
105k_list = log2(M_list);
106Rb = Rs .* k_list;                   % Rb = Rs * log2(M)
107
108T = table(schemes', M_list', k_list', Rb'/1e6, ...
109    'VariableNames', {'Scheme', 'M', 'BitsPerSymbol', 'BitRate_Mbps'});
110fprintf('\nData rate at Rs = %g Msymbol/s:\n', Rs/1e6);
111disp(T);
112
113figure('Name', 'Data Rate');
114bar(Rb/1e6);
115set(gca, 'XTickLabel', schemes);
116ylabel('Bit rate (Mbps)');
117title(sprintf('Data Rate vs Modulation Order (Rs = %g Msym/s)', Rs/1e6));
118grid on;
119
120%% ---------- Helper functions (must stay at the end of the file) ----------
121function y = modulateBits(b, M)
122    if M == 16
123        y = qammod(b, M, 'InputType', 'bit', 'UnitAveragePower', true);
124    else
125        y = pskmod(b, M, (M == 4)*pi/4, 'gray', 'InputType', 'bit');
126    end
127end
128
129function b = demodulateBits(y, M)
130    if M == 16
131        b = qamdemod(y, M, 'OutputType', 'bit', 'UnitAveragePower', true);
132    else
133        b = pskdemod(y, M, (M == 4)*pi/4, 'gray', 'OutputType', 'bit');
134    end
135end
